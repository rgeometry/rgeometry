9/// Check if the given polygon is monotone with respect to given direction
10pub fn is_monotone<T>(poly: &Polygon<T>, direction: &Vector<T, 2>) -> bool
11where
12  T: PolygonScalar,
13{
14  // We can only check polygons without. It would be nice to enforce this with types.
15  assert_eq!(poly.rings.len(), 1);
16
17  let cmp_cursors =
18    |a: &Cursor<'_, T>, b: &Cursor<'_, T>| direction.cmp_along(a, b).then_with(|| a.total_cmp(b));
19  // XXX: Is there a way to get both the min and max element at the same time?
20  let max_cursor = {
21    match poly.iter_boundary().max_by(cmp_cursors) {
22      Some(c) => c,
23      None => return false,
24    }
25  };
26  let min_cursor = {
27    match poly.iter_boundary().min_by(cmp_cursors) {
28      Some(c) => c,
29      None => return false,
30    }
31  };
32
33  // All points going counter-clockwise from min_cursor to max_cursor must be
34  // less-than or equal to the next point in the chain along the direction vector.
35  for pt in min_cursor.to(Excluded(max_cursor)) {
36    if direction.cmp_along(&pt, &pt.next()) == Ordering::Greater {
37      return false;
38    }
39  }
40
41  // Walking down the other chain, the condition is opposite: All points
42  // must be greater-than or equal to the next point in the chain along the direction vector.
43  for pt in max_cursor.to(Excluded(min_cursor)) {
44    if direction.cmp_along(&pt, &pt.next()) == Ordering::Less {
45      return false;
46    }
47  }
48
49  true
50}

52/// Generates a monotone polygon from given points with respect to given direction
53pub fn new_monotone_polygon<T>(
54  mut points: Vec<Point<T, 2>>,
55  direction: &Vector<T, 2>,
56) -> Result<Polygon<T>, Error>
57where
58  T: PolygonScalar,
59{
60  // First compare along the direction vector.
61  // If two points are the same distance along the vector, compare their X and Y components.
62  points.sort_by(|prev, curr| {
63    direction
64      .cmp_along(prev, curr)
65      .then_with(|| prev.total_cmp(curr))
66  });
67
68  points.dedup();
69  if points.len() < 3 {
70    return Err(Error::InsufficientVertices);
71  }
72
73  let (min_point, max_point) = (
74    points.first().unwrap().clone(),
75    points.last().unwrap().clone(),
76  );
77
78  // The polygon is made up of two chains running from 'min_point' to
79  // 'max_point': One on the clockwise side of the line between them and one on
80  // the counter-clockwise side. Points that lie exactly on that line may go on
81  // either chain with one caveat: A chain without any points of its own is just
82  // the straight line from 'min_point' to 'max_point', and a co-linear point on
83  // the other chain would then sit on top of that line, making the polygon
84  // self-intersecting. So, co-linear points are always placed on the chain that
85  // has no points of its own.
86  let last = points.len() - 1;
87  let colinear_side = if points[1..last]
88    .iter()
89    .any(|pt| Orientation::new(&min_point, &max_point, pt) == Orientation::CounterClockWise)
90  {
91    Orientation::ClockWise
92  } else {
93    Orientation::CounterClockWise
94  };
95
96  let mut polygon_points: VecDeque<Point<T, 2>> = VecDeque::new();
97
98  while let Some(curr) = points.pop() {
99    let side = match Orientation::new(&min_point, &max_point, &curr) {
100      Orientation::CoLinear => colinear_side,
101      side => side,
102    };
103    match side {
104      Orientation::ClockWise => polygon_points.push_front(curr),
105      _ => polygon_points.push_back(curr),
106    }
107  }
108  let vec = Vec::from(polygon_points);
109
110  Polygon::new(vec)
111}

143  #[test]

145    let polygon = Polygon::new(vec![
146      Point::new([0, 3]),
147      Point::new([1, 2]),
148      Point::new([1, -2]),
149      Point::new([0, -3]),

155  // Regression test: Points that lie on the line between the extreme points
156  // used to be placed on the same chain as (0,0), leaving the other chain as a
157  // bare line from (113,-8) to (23,127) with (111,-5) sitting on top of it.
158  #[test]
159  fn colinear_points_on_empty_chain() {
160    let points = vec![
161      Point::new([111, -5]),
162      Point::new([23, 127]),
163      Point::new([0, 0]),
164      Point::new([113, -8]),
165    ];
166    let direction = Vector([0, 1]);
167    let polygon = new_monotone_polygon(points, &direction).unwrap();
168    assert!(is_monotone(&polygon, &direction));
169    assert_eq!(polygon.validate().err(), None);
170  }

173  fn monotone_is_monotone_prop(points: Vec<Point<i8, 2>>, direction: Vector<i8, 2>) {
174    if let Ok(p) = new_monotone_polygon(points, &direction) {
175      prop_assert!(is_monotone(&p, &direction));

180  #[proptest]
181  fn valid_monotone(points: Vec<Point<i8, 2>>, direction: Vector<i8, 2>) {
182    // dedup points
183    let mut points = points;
184    let mut set = BTreeSet::new();
185    points.retain(|pt| set.insert(*pt));
186    // If we have at least three, non-colinear points, then we must be able to
187    // create a monotone polygon.
188    if !points
189      .windows(3)
190      .all(|window| Orientation::new(&window[0], &window[1], &window[2]).is_colinear())
191    {
192      let p = new_monotone_polygon(points, &direction).unwrap();
193      prop_assert!(is_monotone(&p, &direction));
194      prop_assert_eq!(p.validate().err(), None);
195    }
196  }
